## Cor Predominante = 
VAR TabelaCor =
    ADDCOLUMNS(
        VALUES(enem2019_basico[cor_raca]),
        "Qtd", CALCULATE([Total Participantes])
    )

VAR TopCor =
    TOPN(1, TabelaCor, [Qtd], DESC)

VAR NomeCor =
    MAXX(TopCor, enem2019_basico[cor_raca])

VAR QtdCor =
    MAXX(TopCor, [Qtd])

VAR Percentual =
    DIVIDE(QtdCor, [Total Participantes])

RETURN
    NomeCor & " (" & FORMAT(Percentual, "0,0%") & ")"
